function timeAgo(d) {
  const s = (Date.now() - new Date(d)) / 1000;
  if (s < 60) return 'เมื่อสักครู่';
  if (s < 3600) return Math.floor(s / 60) + ' นาทีที่แล้ว';
  if (s < 86400) return Math.floor(s / 3600) + ' ชั่วโมงที่แล้ว';
  return Math.floor(s / 86400) + ' วันที่แล้ว';
}

function toast(msg, type = 'ok') {
  const t = document.createElement('div');

function chart(labels, series) {
  const max = Math.max(1, ...series.flatMap(s => s.values));
  return `<div class="chart">${labels.map((lb, i) => `<div class="chart-col"><div class="bars">${series.map(s => `<div class="bar" style="height:${(s.values[i] / max) * 100}%;background:${s.color}" title="${s.name}: ${s.values[i]}"></div>`).join('')}</div><span class="chart-label">${lb}</span></div>`).join('')}</div>
  ${series.length > 1 ? `<div class="legend">${series.map(s => `<span><i style="background:${s.color}"></i>${s.name}</span>`).join('')}</div>` : ''}`;
}
function calcStreak(acts) {
  const days = new Set(acts.map(a => dayKey(a.created_at)));
  const d = new Date(); let n = 0;
  if (!days.has(dayKey(d))) d.setDate(d.getDate() - 1);
  while (days.has(dayKey(d))) { n++; d.setDate(d.getDate() - 1); }
  return n;
}
function calcProgress(V, G, K) {
  const total = V.length + G.length + K.length;
  const learned = V.filter(x => x.interval_days >= 3).length + G.filter(x => x.studied).length + K.filter(x => x.interval_days >= 3).length;
  return total ? Math.round((learned / total) * 100) : 0;
}

async function loadSettings() {
  let { data, error } = await sb.from('user_settings').select('*').maybeSingle();
  if (error) throw error;
  if (!data) {
    const ins = await sb.from('user_settings').insert({ current_language: S.lang, display_name: S.user.email.split('@')[0], goals: { en: DEFAULT_GOALS, ja: DEFAULT_GOALS } }).select().single();
    if (ins.error) throw ins.error;
    data = ins.data;
    const rows = [];
    for (const lang of ['en', 'ja']) {
      DEFAULT_CATS.vocab.forEach(name => rows.push({ language: lang, kind: 'vocab', name }));
      DEFAULT_CATS.grammar[lang].forEach(name => rows.push({ language: lang, kind: 'grammar', name }));
    }
    await sb.from('categories').insert(rows);
  }
  S.settings = data;
  S.lang = data.current_language;
  localStorage.setItem('hub_lang', S.lang);
  await loadCats();
}
async function loadCats() { const { data } = await sb.from('categories').select('*').order('name'); S.cats = data || []; }
const catsOf = kind => S.cats.filter(c => c.language === S.lang && c.kind === kind);
async function ensureCat(kind, name) {
  if (!name || catsOf(kind).some(c => c.name === name)) return;
  const { data, error } = await sb.from('categories').insert({ language: S.lang, kind, name }).select().single();
  if (error) return fail(error);
  S.cats.push(data);
}

async function router() {
  if (!S.user) return;
  S.cleanup?.(); S.cleanup = null;
  const [path, qs = ''] = location.hash.replace(/^#\/?/, '').split('?');
  const parts = path.split('/');
  const page = parts[0] || 'dashboard', param = parts[1];
  if (!ROUTES[page] || (page === 'kanji' && !isJa())) { location.hash = '#/dashboard'; return; }
  renderSidebar(); closeSidebar();
  $('#main').classList.toggle('full', page === 'docs' && !!param);
  if (page !== 'search') $('#global-search').value = '';
  // Each route renders into its own root -> stale async renders never overwrite the new page
  const root = document.createElement('div');
  root.className = 'route';
  root.innerHTML = loader();
  view().replaceChildren(root);
  window.scrollTo(0, 0);
  try { await ROUTES[page](root, param, new URLSearchParams(qs)); }
  catch (e) {
    console.error(e);
    root.innerHTML = `<div class="error-box"><i data-lucide="triangle-alert"></i><p>เกิดข้อผิดพลาด: ${esc(e.message || e)}</p><button class="btn" onclick="location.reload()">ลองใหม่</button></div>`;
  }
  icons();
}
window.addEventListener('hashchange', router);

const titleOf = (key, it) => (key === 'vocab' && it.kanji ? it.kanji : it[COL[key].titleKey]);
function readingOf(key, it) {
  if (key === 'vocab') return isJa()
    ? [...new Set([it.kanji ? it.word : '', it.hiragana, it.katakana, it.romaji].filter(x => x && x !== titleOf('vocab', it)))].join(' · ')
    : [it.ipa || it.pronunciation, it.part_of_speech].filter(Boolean).join(' · ');
  if (key === 'kanji') return [it.onyomi && `音 ${it.onyomi}`, it.kunyomi && `訓 ${it.kunyomi}`].filter(Boolean).join('   ');
  return it.romaji || '';
}

function speak(text) {
  if (!('speechSynthesis' in window)) return toast('เบราว์เซอร์นี้ไม่รองรับการอ่านออกเสียง', 'err');
  speechSynthesis.cancel();
  const u = new SpeechSynthesisUtterance(plain(text));
  u.lang = isJa() ? 'ja-JP' : 'en-US'; u.rate = 0.9;
  speechSynthesis.speak(u);
}

/* ================= ADD / EDIT FORM ================= */
function fieldHtml(f, val, catKind) {
  const v = val ?? '', id = `fld-${f.k}`, req = f.req ? 'required' : '';
  let input;
  if (f.t === 'area') input = `<textarea id="${id}" name="${f.k}" rows="3" placeholder="${esc(f.ph || '')}" ${req}>${esc(v)}</textarea>`;
  else if (f.t === 'select') {
    const o = typeof f.o === 'function' ? f.o() : f.o;
    input = `<select id="${id}" name="${f.k}"><option value="">—</option>${o.map(x => `<option ${x === v ? 'selected' : ''}>${esc(x)}</option>`).join('')}</select>`;
  } else if (f.t === 'cat') {
    input = `<input id="${id}" name="${f.k}" list="dl-${f.k}" value="${esc(v)}" placeholder="เลือก หรือพิมพ์เพื่อสร้างหมวดหมู่ใหม่"><datalist id="dl-${f.k}">${catsOf(catKind).map(x => `<option value="${esc(x.name)}">`).join('')}</datalist>`;
  } else input = `<input id="${id}" name="${f.k}" type="${f.t === 'number' ? 'number' : 'text'}" ${f.t === 'number' ? 'min="1"' : ''} value="${esc(v)}" placeholder="${esc(f.ph || '')}" ${req}>`;
  return `<div class="field ${f.full ? 'full' : ''}"><label for="${id}">${f.l}${f.req ? ' *' : ''}</label>${input}${f.hint ? `<small class="hint">${f.hint}</small>` : ''}</div>`;
}

function openForm(key, item = null) {
  const c = COL[key], fs = visibleFields(c);
  const m = modal(`${item ? 'แก้ไข' : 'เพิ่ม'} ${c.name} ${isJa() ? '🇯🇵' : '🇬🇧'}`,
    `<form id="item-form" class="form-grid">${fs.map(f => fieldHtml(f, item?.[f.k], c.catKind)).join('')}
     <div class="form-actions full"><button type="button" class="btn ghost" data-close>ยกเลิก</button><button class="btn" type="submit"><i data-lucide="save"></i>บันทึก</button></div></form>`, { wide: true });
  const form = $('#item-form', m);
  form.querySelector('input,textarea')?.focus();
  form.onsubmit = async e => {
    e.preventDefault();
    const btn = form.querySelector('[type=submit]');
    btn.disabled = true; btn.textContent = 'กำลังบันทึก...';
    const fd = new FormData(form), row = {};
    fs.forEach(f => { const v = String(fd.get(f.k) ?? '').trim(); row[f.k] = f.t === 'number' ? (v ? parseInt(v, 10) : null) : (v || null); });
    try {
      if (c.catKind && row.category) await ensureCat(c.catKind, row.category);
      const res = item
        ? await sb.from(c.table).update(row).eq('id', item.id).select().single()
        : await sb.from(c.table).insert({ ...row, language: S.lang }).select().single();
      if (res.error) throw res.error;
      if (!item) logActivity(c.addAct);
      closeModal();
      toast(item ? 'บันทึกการแก้ไขแล้ว' : 'เพิ่มรายการเรียบร้อย ✨');
      router();
    } catch (err) {
      fail(err, 'บันทึกไม่สำเร็จ:');
      btn.disabled = false; btn.textContent = 'บันทึก';
    }
  };
}

async function downloadDoc(doc) {
  const { data, error } = await sb.storage.from('documents').createSignedUrl(doc.file_path, 60, { download: doc.file_name });
  if (error) return fail(error, 'ดาวน์โหลดไม่สำเร็จ:');
  const a = document.createElement('a'); a.href = data.signedUrl; a.rel = 'noopener'; document.body.append(a); a.click(); a.remove();
}
async function deleteDoc(doc) {
  if (!(await confirmBox(`ลบเอกสาร "${doc.title}" ใช่หรือไม่? ไฟล์จะถูกลบถาวร`))) return;
  const st = await sb.storage.from('documents').remove([doc.file_path]);
  if (st.error) return fail(st.error, 'ลบไฟล์ไม่สำเร็จ:');
  const { error } = await sb.from('documents').delete().eq('id', doc.id);
  if (error) return fail(error);
  toast('ลบเอกสารแล้ว'); router();
}

        <button class="icon-btn" id="v-dl" title="ดาวน์โหลด"><i data-lucide="download"></i></button>
      </div>
    </div>
    <div class="viewer-scroll" id="v-scroll">${loader('กำลังเปิดเอกสาร...')}</div>
  </div>`;
  icons();
  $('#v-dl', root).onclick = () => downloadDoc(doc);

  const pdfjs = await loadPdfJs();
  const signed = await sb.storage.from('documents').createSignedUrl(doc.file_path, 3600);
  if (signed.error) throw signed.error;
  const pdf = await pdfjs.getDocument({ url: signed.data.signedUrl }).promise;

  const scroller = $('#v-scroll', root), N = pdf.numPages;
  $('#v-total', root).textContent = N; $('#v-page', root).max = N;
  const base = (await pdf.getPage(1)).getViewport({ scale: 1 });
  const fitScale = () => Math.min(2, Math.max(0.4, (scroller.clientWidth - 32) / base.width));
  const dpr = Math.min(window.devicePixelRatio || 1, 2);
  let scale = fitScale(), gen = 0, current = 1, observer = null;

  const renderPage = async (el, myGen) => {
    if (el.dataset.done) return;
    el.dataset.done = '1';
    const page = await pdf.getPage(+el.dataset.page);
    if (myGen !== gen) return;
    const vp = page.getViewport({ scale: scale * dpr });
    const canvas = document.createElement('canvas');
    canvas.width = vp.width; canvas.height = vp.height;
    canvas.style.width = vp.width / dpr + 'px'; canvas.style.height = vp.height / dpr + 'px';
    await page.render({ canvasContext: canvas.getContext('2d'), viewport: vp }).promise;
    if (myGen !== gen) return;
    el.style.width = canvas.style.width; el.style.height = canvas.style.height;
    el.replaceChildren(canvas);
  };
  const build = () => {
    const myGen = ++gen;
    observer?.disconnect();
    scroller.innerHTML = '';
    for (let i = 1; i <= N; i++) {
      const d = document.createElement('div');
      d.className = 'pdf-page'; d.dataset.page = i;
      d.style.width = base.width * scale + 'px'; d.style.height = base.height * scale + 'px';
      scroller.append(d);
    }
    observer = new IntersectionObserver(ents => ents.forEach(en => en.isIntersecting && renderPage(en.target, myGen)), { root: scroller, rootMargin: '800px 0px' });
    $('.pdf-page', scroller).forEach(p => observer.observe(p));
    $('#v-fit', root).textContent = Math.round(scale * 100) + '%';
  };
  const setCur = n => { current = n; $('#v-page', root).value = n; };
  const pageEl = n => scroller.querySelector(`[data-page="${n}"]`);
  const goTo = n => { n = Math.max(1, Math.min(N, n || 1)); scroller.scrollTo({ top: pageEl(n).offsetTop - 12, behavior: 'smooth' }); setCur(n); };
  const zoom = s => {
    const keep = current;
    scale = Math.min(4, Math.max(0.3, s));
    build();
    requestAnimationFrame(() => { scroller.scrollTop = pageEl(keep).offsetTop - 12; setCur(keep); });
  };
  scroller.addEventListener('scroll', () => {
    const mid = scroller.scrollTop + scroller.clientHeight / 3;
    let n = 1;
    for (const p of scroller.children) { if (p.offsetTop <= mid) n = +p.dataset.page; else break; }
    if (n !== current) setCur(n);
  }, { passive: true });

  $('#v-prev', root).onclick = () => goTo(current - 1);
  $('#v-next', root).onclick = () => goTo(current + 1);
  $('#v-page', root).onchange = e => goTo(+e.target.value);
  $('#v-in', root).onclick = () => zoom(scale + 0.2);
  $('#v-out', root).onclick = () => zoom(scale - 0.2);
  $('#v-fit', root).onclick = () => zoom(fitScale());
  $('#v-full', root).onclick = () => {
    const v = $('#viewer', root);
    if (document.fullscreenElement) document.exitFullscreen();
    else if (v.requestFullscreen) v.requestFullscreen().catch(() => v.classList.toggle('pseudo-full'));
    else v.classList.toggle('pseudo-full'); // iOS Safari fallback
  };
  const onKey = e => {
    if (e.target.tagName === 'INPUT') return;
    if (['ArrowRight', 'PageDown'].includes(e.key)) goTo(current + 1);
    if (['ArrowLeft', 'PageUp'].includes(e.key)) goTo(current - 1);
    if (e.key === 'Escape') $('#viewer', root)?.classList.remove('pseudo-full');
  };
  document.addEventListener('keydown', onKey);
  S.cleanup = () => { document.removeEventListener('keydown', onKey); observer?.disconnect(); pdf.destroy(); };
  build();
}

const GRADES = [{ th: 'จำไม่ได้', en: 'Again' }, { th: 'จำได้ยาก', en: 'Hard' }, { th: 'จำได้', en: 'Good' }, { th: 'จำได้ดี', en: 'Easy' }];
function schedule(c, g) {
  let ease = c.ease ?? 2.5, iv = c.interval_days ?? 0, reps = c.reps ?? 0;
  const now = new Date(); let due;
  if (g === 0) { reps = 0; iv = 0; ease = Math.max(1.3, ease - 0.2); due = new Date(now.getTime() + 10 * 60000); }
  else {
    if (g === 1) { iv = Math.max(1, Math.round(iv * 1.2)); ease = Math.max(1.3, ease - 0.15); }
    else if (g === 2) { iv = reps === 0 ? 1 : reps === 1 ? 3 : Math.round(iv * ease); }
    else { iv = reps === 0 ? 4 : Math.round(Math.max(iv, 1) * ease * 1.3); ease += 0.15; }
    reps++; due = new Date(now.getTime() + iv * 864e5);
  }
  return { ease: +ease.toFixed(2), interval_days: iv, reps, due_at: due.toISOString(), last_studied_at: now.toISOString() };
}
const previewInterval = (c, g) => (g === 0 ? '10 นาที' : schedule(c, g).interval_days + ' วัน');

function buildMC(pool, promptFn, answerFn, subFn, label) {
  const valid = pool.filter(answerFn);
  const answers = [...new Set(valid.map(answerFn))];
  if (answers.length < 4) return [];
  return shuffle(valid).slice(0, 10).map(x => {
    const a = answerFn(x);
    return { prompt: promptFn(x), sub: subFn(x), label, answer: a, choices: shuffle([a, ...shuffle(answers.filter(y => y !== a)).slice(0, 3)]) };
  });
}
function buildFill(V) {
  return shuffle(V.filter(v => v.example && [v.word, v.kanji].some(w => w && v.example.toLowerCase().includes(w.toLowerCase())))).slice(0, 10).map(v => {
    const w = [v.kanji, v.word].find(x => x && v.example.toLowerCase().includes(x.toLowerCase()));
    const re = new RegExp(w.replace(/[.*+?^${}()|[\]\\]/g, '\\ CODEBLOCK5 '), 'gi');
    return { sentence: v.example.replace(re, '_____'), answer: w, accept: [v.word, v.kanji, v.hiragana, v.romaji].filter(Boolean).map(s => s.toLowerCase()), hint: v.meaning_th, tr: v.example_translation };
  });
}

  const queue = [...cards], seen = new Set(), total = cards.length, t0 = Date.now();
  let done = 0, remembered = 0, flipped = false, finished = false;
  const onKey = e => {
    if (e.code === 'Space') { e.preventDefault(); flip(); }
    else if (flipped && ['1', '2', '3', '4'].includes(e.key)) grade(+e.key - 1);
  };
  const finish = () => {
    if (finished) return; finished = true;
    document.removeEventListener('keydown', onKey);
    saveResult(root, title, remembered, done, t0, 'review', again);
  };
  const flip = () => {
    const fc = $('#fc', root); if (!fc) return;
    flipped = !flipped; fc.classList.toggle('flipped', flipped);
    if (flipped) $('#grades', root).hidden = false;
  };
  const grade = g => {
    const c = queue.shift();
    const upd = schedule(c, g);
    sb.from(c._t).update(upd).eq('id', c.id).then(({ error }) => error && fail(error));
    Object.assign(c, upd);
    if (!seen.has(c.id)) { seen.add(c.id); done++; if (g >= 2) remembered++; }
    if (g === 0) queue.push(c); // จำไม่ได้ -> วนกลับมาท้ายรอบนี้
    show();
  };
  const show = () => {
    if (!queue.length) return finish();
    const c = queue[0]; flipped = false;
    root.innerHTML = sessionHead(title, done, total) + `
      <div class="flashcard" id="fc"><div class="fc-inner">
        <div class="fc-face fc-front">${fcFront(c)}<span class="fc-hint">แตะเพื่อพลิกการ์ด</span></div>
        <div class="fc-face fc-back">${fcBack(c)}</div>
      </div></div>
      <div class="grades" id="grades" hidden>${GRADES.map((gr, i) => `<button class="grade g${i}" data-g="${i}"><b>${gr.th}</b><small>${gr.en} · ${previewInterval(c, i)}</small></button>`).join('')}</div>
      <p class="muted center small">Space = พลิกการ์ด · 1–4 = ให้คะแนนความจำ</p></div>`;
    icons();
    $('#fc', root).onclick = flip;
    $('#grades', root).onclick = e => { const b = e.target.closest('[data-g]'); if (b) grade(+b.dataset.g); };
    $('#quit', root).onclick = finish;
  };
  document.addEventListener('keydown', onKey);
  S.cleanup = () => document.removeEventListener('keydown', onKey);
  show();
}

function startQuiz(root, title, qs, again) {

    const q = qs[i]; let answered = false;

      <p class="muted small">${esc(q.label)}</p><h2 class="quiz-prompt jp">${rich(q.prompt)}</h2>

      if (q.choices[+b.dataset.j] === q.answer) score++;
      $('.choice', root).forEach(x => {
        const v = q.choices[+x.dataset.j];
        if (v === q.answer) x.classList.add('correct'); else if (x === b) x.classList.add('wrong');

}

function startMatching(root, V, again) {
  const set = shuffle(V.filter(x => x.meaning_th)).slice(0, 6);
  if (set.length < 4) return toast('ต้องมีคำศัพท์อย่างน้อย 4 คำ', 'err');
  const t0 = Date.now(); let L = null, R = null, matched = 0, mistakes = 0;
  root.innerHTML = sessionHead('Matching Words', 0, set.length) + `
    <p class="muted center small">แตะคำศัพท์ แล้วแตะความหมายที่ตรงกัน</p>
    <div class="match-grid">
      <div class="match-col">${set.map(v => `<button class="match-item jp" data-id="${v.id}" data-side="L">${esc(v.kanji || v.word)}</button>`).join('')}</div>
      <div class="match-col">${shuffle(set).map(v => `<button class="match-item" data-id="${v.id}" data-side="R">${esc(v.meaning_th)}</button>`).join('')}</div>
    </div></div>`;
  icons();
  $('#quit', root).onclick = () => router();
  $('.match-grid', root).onclick = e => {
    const b = e.target.closest('.match-item'); if (!b || b.classList.contains('matched')) return;
    if (b.dataset.side === 'L') { L?.classList.remove('sel'); L = b; } else { R?.classList.remove('sel'); R = b; }
    b.classList.add('sel');
    if (!(L && R)) return;
    if (L.dataset.id === R.dataset.id) {
      [L, R].forEach(x => { x.classList.remove('sel'); x.classList.add('matched'); x.disabled = true; });
      matched++;
      $('.pill', root).textContent = `${matched}/${set.length}`;
      $('.progress div', root).style.width = (matched / set.length) * 100 + '%';
      if (matched === set.length) setTimeout(() => saveResult(root, 'Matching Words', Math.max(0, set.length - mistakes), set.length, t0, 'quiz', again), 400);
    } else {
      mistakes++;
      [L, R].forEach(x => { x.classList.remove('sel'); x.classList.add('bad'); setTimeout(() => x.classList.remove('bad'), 400); });
    }
    L = R = null;
  };
}

    const b = e.target.closest('[data-type]'); if (!b) return;
    type = b.dataset.type; $('.fchip', root).forEach(x => x.classList.toggle('on', x === b)); draw();
  };

  draw();

  btn.disabled = true; msg.className = 'auth-msg'; msg.textContent = '';

    ? await sb.auth.signInWithPassword({ email, password })
    : await sb.auth.signUp({ email, password });
  btn.disabled = false;
  if (error) { msg.textContent = error.message; return; }
  if (authMode === 'up' && !data.session) { msg.className = 'auth-msg ok'; msg.textContent = 'สมัครสำเร็จ! กรุณายืนยันอีเมลก่อนเข้าสู่ระบบ'; }
};

async function start(session) {
  const user = session?.user || null;
  if (user && S.user?.id === user.id) return;
  S.user = user;
  if (!user) { S.settings = null; $('#app').hidden = true; $('#auth').hidden = false; view().innerHTML = ''; return; }
  $('#auth').hidden = true; $('#app').hidden = false;
  applyLang();
  view().innerHTML = loader('กำลังเตรียมพื้นที่เรียน...');
  try { await loadSettings(); } catch (e) { fail(e, 'โหลดการตั้งค่าไม่สำเร็จ:'); }
  applyLang();
  if (!location.hash || location.hash === '#' || location.hash === '#/') location.hash = '#/dashboard'; else router();
}

if (SUPABASE_URL.includes('YOUR-')) {
  $('#auth').hidden = false;
  $('#auth-msg').textContent = '⚠️ กรุณาใส่ SUPABASE_URL และ SUPABASE_ANON_KEY ในไฟล์ config.js ก่อนใช้งาน';
} else {
  sb.auth.getSession().then(({ data }) => start(data.session));
  sb.auth.onAuthStateChange((event, session) => {
    if (event === 'SIGNED_IN') start(session);
    if (event === 'SIGNED_OUT') start(null);
  });
}
